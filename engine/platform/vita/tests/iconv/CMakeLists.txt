cmake_minimum_required(VERSION 3.22)
project(VitaIconvContract LANGUAGES CXX)
if(NOT VITASDK)
  message(FATAL_ERROR "Use VitaSDK toolchain")
endif()
set(Iconv_IS_BUILT_IN FALSE CACHE BOOL "" FORCE)
set(Iconv_INCLUDE_DIR "${ODH_VITA_ICONV_ROOT}/include" CACHE PATH "" FORCE)
set(Iconv_LIBRARY "${ODH_VITA_ICONV_ROOT}/lib/libiconv.a" CACHE FILEPATH "" FORCE)
find_package(Iconv REQUIRED)
get_filename_component(ODH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_executable(vita_iconv_contract iconv_probe.cpp "${ODH_ROOT}/tools/linux_text_codec.cpp")
target_include_directories(vita_iconv_contract PRIVATE "${ODH_ROOT}/tools")
target_compile_features(vita_iconv_contract PRIVATE cxx_std_17)
target_link_libraries(vita_iconv_contract PRIVATE Iconv::Iconv)
target_link_options(vita_iconv_contract PRIVATE -Wl,--no-undefined)
