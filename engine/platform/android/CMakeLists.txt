cmake_minimum_required(VERSION 3.22)
project(OpenDollhouseAndroid LANGUAGES C CXX)
if(NOT ANDROID)
  message(FATAL_ERROR "Use the Android NDK toolchain")
endif()
if(NOT EXISTS "${ODH_SDL_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Set ODH_SDL_SOURCE to the official SDL2 source directory")
endif()
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
# The game has no sensor consumer. SDL2 2.30 sensor code uses an API removed in NDK 30.
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
add_subdirectory("${ODH_SDL_SOURCE}" "${CMAKE_BINARY_DIR}/SDL2")
set(ODH_BUILD_DESKTOP OFF CACHE BOOL "" FORCE)
set(ODH_BUILD_LEGACY_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Bionic iconv is API28+. Keep API26 by linking the replaceable GNU library.
set(ODH_ANDROID_ICONV_ROOT "" CACHE PATH "API26 libiconv prefix from prepare-android-iconv.sh")
if(NOT EXISTS "${ODH_ANDROID_ICONV_ROOT}/include/iconv.h" OR
   NOT EXISTS "${ODH_ANDROID_ICONV_ROOT}/lib/libiconv.so")
  message(FATAL_ERROR "Prepare Android libiconv and set ODH_ANDROID_ICONV_ROOT; system iconv requires API28")
endif()
set(Iconv_IS_BUILT_IN FALSE CACHE BOOL "Use packaged API26 iconv" FORCE)
set(Iconv_INCLUDE_DIR "${ODH_ANDROID_ICONV_ROOT}/include" CACHE PATH "Android iconv header" FORCE)
set(Iconv_LIBRARY "${ODH_ANDROID_ICONV_ROOT}/lib/libiconv.so" CACHE FILEPATH "Android iconv library" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../.." "${CMAKE_BINARY_DIR}/engine-root")
add_library(main SHARED AndroidMain.cpp ../desktop/SdlAudio.cpp
  "${CMAKE_CURRENT_LIST_DIR}/../../../tools/reference_patch_setup.cpp")
target_compile_features(main PRIVATE cxx_std_17)
target_link_libraries(main PRIVATE odh_engine SDL2::SDL2)
