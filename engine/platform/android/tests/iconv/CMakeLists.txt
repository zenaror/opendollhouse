cmake_minimum_required(VERSION 3.22)
project(AndroidIconvContract LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_PLATFORM STREQUAL "android-26")
  message(FATAL_ERROR "This link contract must target Android API26")
endif()
set(Iconv_IS_BUILT_IN FALSE CACHE BOOL "" FORCE)
set(Iconv_INCLUDE_DIR "${ODH_ANDROID_ICONV_ROOT}/include" CACHE PATH "" FORCE)
set(Iconv_LIBRARY "${ODH_ANDROID_ICONV_ROOT}/lib/libiconv.so" CACHE FILEPATH "" FORCE)
find_package(Iconv REQUIRED)
get_filename_component(ODH_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_library(android_iconv_contract SHARED iconv_abi_probe.cpp "${ODH_ROOT}/tools/linux_text_codec.cpp")
target_include_directories(android_iconv_contract PRIVATE "${ODH_ROOT}/tools")
target_compile_features(android_iconv_contract PRIVATE cxx_std_17)
target_link_libraries(android_iconv_contract PRIVATE Iconv::Iconv)
target_link_options(android_iconv_contract PRIVATE -Wl,--no-undefined)
