cmake_minimum_required(VERSION 3.22)
project(OpenDollhouseSwitch VERSION 0.0.1 LANGUAGES C CXX)
if(NOT NINTENDO_SWITCH)
  message(FATAL_ERROR "Use devkitPro's Switch.cmake toolchain")
endif()
set(ODH_BUILD_DESKTOP OFF CACHE BOOL "" FORCE)
set(ODH_BUILD_LEGACY_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ODH_SWITCH_ICONV_ROOT "" CACHE PATH "Local static libiconv prefix from prepare-switch-iconv.sh")
if(NOT EXISTS "${ODH_SWITCH_ICONV_ROOT}/include/iconv.h" OR
   NOT EXISTS "${ODH_SWITCH_ICONV_ROOT}/lib/libiconv.a")
  message(FATAL_ERROR "Prepare Switch iconv and set ODH_SWITCH_ICONV_ROOT; SDK headers alone are insufficient")
endif()
set(Iconv_INCLUDE_DIR "${ODH_SWITCH_ICONV_ROOT}/include" CACHE PATH "Switch iconv header" FORCE)
set(Iconv_LIBRARY "${ODH_SWITCH_ICONV_ROOT}/lib/libiconv.a" CACHE FILEPATH "Switch iconv library" FORCE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ODH_SWITCH_SDL REQUIRED IMPORTED_TARGET sdl2>=2.0.18)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../.." "${CMAKE_BINARY_DIR}/engine-root")
add_executable(opendollhouse SwitchMain.cpp ../desktop/SdlAudio.cpp
  "${CMAKE_CURRENT_LIST_DIR}/../../../tools/reference_patch_setup.cpp")
target_compile_features(opendollhouse PRIVATE cxx_std_17)
target_compile_definitions(opendollhouse PRIVATE SDL_MAIN_HANDLED ODH_PLATFORM_SWITCH)
target_link_libraries(opendollhouse PRIVATE odh_engine PkgConfig::ODH_SWITCH_SDL)
nx_generate_nacp(OUTPUT "${CMAKE_BINARY_DIR}/opendollhouse.nacp" NAME "OpenDollhouse" AUTHOR "OpenDollhouse contributors" VERSION "0.0.1")
nx_create_nro(opendollhouse NACP "${CMAKE_BINARY_DIR}/opendollhouse.nacp" NOICON)
