cmake_minimum_required(VERSION 3.16)
project(OpenDollhouse VERSION 0.0.1 LANGUAGES CXX)

add_library(odh_formats src/formats.cpp src/ttab.cpp src/slot.cpp)
target_include_directories(odh_formats PUBLIC include)
target_compile_features(odh_formats PUBLIC cxx_std_17)
set_target_properties(odh_formats PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(odh_formats PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

option(ODH_SANITIZERS "Enable address and undefined-behavior sanitizers on host" OFF)
if(ODH_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "ODH_SANITIZERS requires GCC or Clang")
  endif()
  target_compile_options(odh_formats PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(odh_formats PUBLIC -fsanitize=address,undefined)
endif()

add_executable(odh-inspect tools/inspect.cpp)
target_link_libraries(odh-inspect PRIVATE odh_formats)
add_executable(odh-ttab tools/ttab.cpp)
target_link_libraries(odh-ttab PRIVATE odh_formats)
add_executable(odh-slot tools/slot.cpp)
target_link_libraries(odh-slot PRIVATE odh_formats)
add_library(odh_vm src/vm.cpp)
target_link_libraries(odh_vm PUBLIC odh_formats)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(odh_vm PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_library(odh_world src/world.cpp)
target_link_libraries(odh_world PUBLIC odh_vm)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(odh_world PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_library(odh_route src/route.cpp)
target_link_libraries(odh_route PUBLIC odh_formats)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(odh_route PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
add_executable(odh-bhav tools/bhav.cpp)
target_link_libraries(odh-bhav PRIVATE odh_world)
include(CTest)
if(BUILD_TESTING)
  add_executable(odh-format-tests tests/formats.cpp)
  target_link_libraries(odh-format-tests PRIVATE odh_formats)
  add_test(NAME formats COMMAND odh-format-tests)
  add_executable(odh-ttab-tests tests/ttab.cpp)
  target_link_libraries(odh-ttab-tests PRIVATE odh_formats)
  add_test(NAME ttab COMMAND odh-ttab-tests)
  add_executable(odh-slot-tests tests/slot.cpp)
  target_link_libraries(odh-slot-tests PRIVATE odh_formats)
  add_test(NAME slot COMMAND odh-slot-tests)
  add_executable(odh-vm-tests tests/vm.cpp)
  target_link_libraries(odh-vm-tests PRIVATE odh_vm)
  add_test(NAME vm COMMAND odh-vm-tests)
  add_executable(odh-service-tests tests/services.cpp)
  target_link_libraries(odh-service-tests PRIVATE odh_vm)
  add_test(NAME services COMMAND odh-service-tests)
  add_executable(odh-world-tests tests/world.cpp)
  target_link_libraries(odh-world-tests PRIVATE odh_world)
  add_test(NAME world COMMAND odh-world-tests)
  add_executable(odh-route-tests tests/route.cpp)
  target_link_libraries(odh-route-tests PRIVATE odh_route)
  add_test(NAME route COMMAND odh-route-tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME inspector_cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py $<TARGET_FILE:odh-inspect>)
    add_test(NAME ttab_cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/ttab_cli.py $<TARGET_FILE:odh-ttab>)
    add_test(NAME slot_cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/slot_cli.py $<TARGET_FILE:odh-slot>)
    add_test(NAME bhav_cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/bhav_cli.py $<TARGET_FILE:odh-bhav>)
  endif()
endif()
