cmake_minimum_required(VERSION 3.16)
project(OpenDollhouse VERSION 0.0.1 LANGUAGES CXX)

add_library(odh_formats src/formats.cpp src/ttab.cpp)
target_include_directories(odh_formats PUBLIC include)
target_compile_features(odh_formats PUBLIC cxx_std_17)
set_target_properties(odh_formats PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(odh_formats PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

option(ODH_SANITIZERS "Enable address and undefined-behavior sanitizers on host" OFF)
if(ODH_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "ODH_SANITIZERS requires GCC or Clang")
  endif()
  target_compile_options(odh_formats PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(odh_formats PUBLIC -fsanitize=address,undefined)
endif()

add_executable(odh-inspect tools/inspect.cpp)
target_link_libraries(odh-inspect PRIVATE odh_formats)
add_executable(odh-ttab tools/ttab.cpp)
target_link_libraries(odh-ttab PRIVATE odh_formats)
include(CTest)
if(BUILD_TESTING)
  add_executable(odh-format-tests tests/formats.cpp)
  target_link_libraries(odh-format-tests PRIVATE odh_formats)
  add_test(NAME formats COMMAND odh-format-tests)
  add_executable(odh-ttab-tests tests/ttab.cpp)
  target_link_libraries(odh-ttab-tests PRIVATE odh_formats)
  add_test(NAME ttab COMMAND odh-ttab-tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME inspector_cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py $<TARGET_FILE:odh-inspect>)
    add_test(NAME ttab_cli COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/ttab_cli.py $<TARGET_FILE:odh-ttab>)
  endif()
endif()
